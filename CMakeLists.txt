cmake_minimum_required(VERSION 3.16)
project(RetroSfxRenderer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(retro_render native/render.cpp vendor/ymfm/src/ymfm_misc.cpp vendor/ymfm/src/ymfm_opl.cpp vendor/ymfm/src/ymfm_opm.cpp vendor/ymfm/src/ymfm_opn.cpp vendor/ymfm/src/ymfm_ssg.cpp vendor/ymfm/src/ymfm_adpcm.cpp vendor/ymfm/src/ymfm_pcm.cpp)
target_include_directories(retro_render PRIVATE vendor/ymfm/src)
if(MSVC)
 target_compile_options(retro_render PRIVATE /O2 /EHsc)
endif()
